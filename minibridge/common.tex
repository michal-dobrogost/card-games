% ---------- SIDE A: RULES ----------
\newcommand{\sideA}{%
  \cardtitle{MiniBridge Rules}
  {\raggedright\begin{enumerate}[leftmargin=1.5em,itemsep=0.5\baselineskip,topsep=0pt,parsep=0pt]
    \item \textbf{Deal} a 52-card deck to four players (13 cards each). Two teams of two. Partners sit opposite.
    \item \textbf{Sum} and announce high-card points (HCP), clockwise, dealer first:
      \\[0.3\baselineskip]
      \hspace*{0.4em}\textbf{A}~=~4\qquad\textbf{K}~=~3\qquad\textbf{Q}~=~2\qquad\textbf{J}~=~1
    \item
      \textbf{Declarer:} the side with more combined HCP declares.
      The partner with more HCP is declarer; if tied, whoever announced first.
      Declarer's partner lays their hand face up as dummy.
    \item \textbf{Contract:} declarer names a level and a trump suit, or no trump.
      The level is the number of tricks needed beyond 6 (e.g.\ 4\Heart{} = 10 tricks).
      Consider trumps with 8+ cards in a suit across both hands (a fit).
      \\[0.3\baselineskip]
      \begin{tabular}[c]{|l|llll|}
        \hline
        \multirow{2}{*}{Contract suit} & \multicolumn{4}{c|}{Combined HCP} \\
        & 21+ & 23+ & 25+ & 27+ \\
        \hline
        No trump                          & 1 & 2 & \textbf{3} & -- \\
        Major (\Spade{} or \Heart{})      & 2 & 3 & \textbf{4} & -- \\
        Minor (\Diamondsuit{} or \Club{}) & 2 & 3 & 4 & \textbf{5} \\
        \hline
      \end{tabular}\quad
      \parbox[c]{8em}{\raggedright level you may expect to make}
    \item
      \textbf{Play:} the player left of declarer leads one card face up; others follow clockwise.
      \emph{You must follow suit if you can}; otherwise play a trump or any other card.
      Declarer plays both their own hand and dummy.
    \item
      \textbf{Tricks} are won by the highest trump, else the highest card of the suit led (A, K, Q, J, 10 ... 2).
      The 4 cards played are turned face down and the winner leads the next trick.
    \item \textbf{Score} the tricks won. Deal passes clockwise.
  \end{enumerate}}

  \cardfooter
}

\newcommand{\cell}[1]{%
  \vbox to \cellh{\vss\hbox to \cellw{\hss
    \parbox[c][\dimexpr\cellh-2\inset\relax][t]{\dimexpr\cellw-2\inset\relax}{\bodysize #1}%
  \hss}\vss}}

\newcommand{\fourup}[1]{%
  \noindent\vbox to \paperheight{\offinterlineskip\cellrow{#1}\cellrow{#1}}}

% Dashed cut lines through the page centre; drawn on the front page only, so the
% back side prints clean. Must follow \fourup in the same paragraph.
\newcommand{\cutlines}{%
  \begin{tikzpicture}[remember picture,overlay]
    \draw[gray!60,dash pattern=on 3pt off 3pt,thin]
      (current page.north) -- (current page.south);
    \draw[gray!60,dash pattern=on 3pt off 3pt,thin]
      (current page.west) -- (current page.east);
  \end{tikzpicture}}

% One cell (laid out exactly as in the 4-up version) scaled 2x to fill the page.
\newcommand{\oneup}[1]{%
  \noindent\vbox to \paperheight{\offinterlineskip\scalebox{2}{\cell{#1}}\vss}}
