% ---------- SIDE A: RULES ----------
\newcommand{\sideA}{%
  \cardtitle{Thousand (1000)}
  {\raggedright
  An excellent three-player trick-taking game with changing trumps.\gap

  \begin{enumerate}[leftmargin=1.5em,itemsep=0.5\baselineskip,topsep=0pt,parsep=0pt]
    \item
      \textbf{Deal} 7 cards each to 3 players, clockwise from dealer's left, and 3 face down as a kitty (24-card deck: A, K, Q, J, 10, 9 per suit).
    \item
      \textbf{Bid} clockwise from dealer's left: pass or raise, minimum 100, in 5 point increments.
      Once you pass, you're out.
      If everyone passes, dealer bids 100.
    \item
      \textbf{Kitty:} the high bidder takes the kitty into their hand, passes 1 card face down to each opponent, and may raise their bid.
    \item
      \textbf{Play:} the high bidder leads.
      \emph{You must follow suit if you can}, otherwise trump if you can, otherwise play anything.
      Highest trump wins, else highest card of the suit led.
      The winner leads next.
    \item
      \textbf{Card rank and points} (120 total):\par\vspace{0.3\baselineskip}
      \hspace*{0.5em}\textbf{A}~11\qquad\textbf{10}~10\qquad\textbf{K}~4\qquad\textbf{Q}~3\qquad\textbf{J}~2\qquad\textbf{9}~0
    \item
      \textbf{Marriage:} when you lead a Q and hold the K of the same suit, you may declare a new marriage.
      There is no trump until a marriage is declared. The suit declared becomes trump, and you score:\par\vspace{0.3\baselineskip}
      \hspace*{0.5em}\Heart~100\qquad\Diamondsuit~80\qquad\Club~60\qquad\Spade~40
    \item
      \textbf{Score} your card points plus marriages.
      The bidder scores their bid if they made it, otherwise minus it.
      First to 1000 wins. Scores can go negative.
    \item
      \textbf{Deal} passes clockwise.
  \end{enumerate}}
}

% ---------- SIDE B: SCORING ----------
\newcommand{\sideB}{%
  \raggedright
  \renewcommand{\gap}{\vspace{0.5\baselineskip}}% tighter than the rules side, so it all fits
  \cardtitle{Makao}

   A shedding game for 2+ players, like uno or crazy eights, from Eastern Europe.
   Pronounced mah-KAH-o \textipa{/ma"ka.O/}.
   \gap

  \begin{enumerate}[leftmargin=1.5em,itemsep=0.5\baselineskip,topsep=0pt,parsep=0pt]
    \item
      \textbf{Deal} 5 cards each to 2+ players, clockwise from dealer's left.
      At the end place one card face up as the discard pile and the deck, face down, becomes the pickup pile.
      A 52-card deck is used with jokers.
    \item
      \textbf{First} to play is the loser of the previous round, else the youngest.
    \item
      \textbf{Play} any card from your hand if it matches the suit or denomination (e.g. 5 or A) of the top of the discard pile.
      If you have multiple cards of the same denomination you may play them in one turn (e.g. 3 aces).
      If you can not play a card, pick one up from the pickup pile; you may immediately play it.
    \item \textbf{Actions} of special cards on the next player:
      2 - skip \& pick up 2
      3 - skip \& pick up 3
      4 - skip a turn
      J - demand a non-action card (each player must place the named card) or demand nothing
      Q - may be played on any suited non-action card and any card may be played on it.
      KH/KS - next/previous picks up 5
      A - name a new suit
      Joker - may turn into any card as named by the player.
    \item
      \textbf{Special cards} may be deflected. E.g. if I play the 2H the next player may play 3H and the player after takes 5 cards.
      KH/KS may be blocked by KD/KC or deflected by the 2 or 3 of the same suit.
      KS may not be defelected in a 3+ person game.
      J may be deflected only if a demanded card has not been already played.
    \item
      \textbf{Makao} must be said when the second last card is played to alert others.
      This may happen in the middle of placing multiple cards in one turn.
      If someone notices you didn't say A penalty of 5 cards applies if the next person plays before you say it and someone notices.
    \item
      \textbf{Winner} is the first to an empty hand.
      You may say "po makale" (paw mah-KAH-leh \textipa{/pO ma"ka.lE/}) as a curtouisy.
  \end{enumerate}

  \gap
  \vfill{\fontsize{5.5}{6.5}\selectfont\noindent\expandafter\url\expandafter{\footurl}}
}

% Dashed cut lines through the page centre; drawn on the front page only, so the
% back side prints clean. Must follow \fourup in the same paragraph.
\newcommand{\cutlines}{%
  \begin{tikzpicture}[remember picture,overlay]
    \draw[gray!60,dash pattern=on 3pt off 3pt,thin]
      (current page.north) -- (current page.south);
    \draw[gray!60,dash pattern=on 3pt off 3pt,thin]
      (current page.west) -- (current page.east);
  \end{tikzpicture}}
