% ---------- SIDE A: THOUSAND ----------
\newcommand{\sideA}{%
  \cardtitle{Thousand (1000)}
  {\raggedright
  An excellent three-player trick-taking game with changing trumps.\gap

  \begin{enumerate}[leftmargin=1.5em,itemsep=0.5\baselineskip,topsep=0pt,parsep=0pt]
    \item
      \textbf{Deal} 7 cards each to 3 players, clockwise from dealer's left, and 3 face down as a kitty (24-card deck: A, K, Q, J, 10, 9 per suit).
    \item
      \textbf{Bid} clockwise from dealer's left: pass or raise, minimum 100, in multiples of 5.
      Once you pass, you're out.
      If everyone passes, dealer bids 100.
    \item
      \textbf{Kitty:} the high bidder takes the kitty into their hand, passes 1 card face down to each opponent, and may raise their bid.
    \item
      \textbf{Play:} the high bidder leads.
      \emph{You must follow suit if you can}, otherwise trump if you can, otherwise play anything.
      Highest trump wins, else highest card of the suit led.
      The winner leads next.
    \item
      \textbf{Card rank and points} (120 total):\par\vspace{0.3\baselineskip}
      \hspace*{0.5em}\textbf{A}~11\qquad\textbf{10}~10\qquad\textbf{K}~4\qquad\textbf{Q}~3\qquad\textbf{J}~2\qquad\textbf{9}~0
    \item
      \textbf{Marriage:} when you lead a Q and hold the K of the same suit, you declare a marriage.
      There is no trump until a marriage is declared. The suit declared becomes trump, and you score:\par\vspace{0.3\baselineskip}
      \hspace*{0.5em}\Heart~100\qquad\Diamondsuit~80\qquad\Club~60\qquad\Spade~40
    \item
      \textbf{Score} your card points plus marriages.
      The bidder scores their bid if they made it, otherwise minus it.
      First to 1000 wins. Scores can go negative.
    \item
      \textbf{Deal} passes clockwise.
  \end{enumerate}}

  \cardfooter{\footurlA}
}

% ---------- SIDE B: MAKAO ----------
% \actleft{...} = left column of a paired action line, padded to \actw so the
% right-hand entries line up (\actw is set inside \sideB, at the body font size).
\newlength{\actw}
\newcommand{\actleft}[1]{\makebox[\actw][l]{#1}\qquad}
\newcommand{\sideB}{%
  \raggedright
  \renewcommand{\gap}{\vspace{0.5\baselineskip}}% tighter than the rules side, so it all fits
  \cardtitle{Makao}

  A shedding game for 2+ players from Eastern Europe, like Uno or Crazy Eights.
  Pronounced mah-KAH-o \textipa{/ma"ka.O/}.
  \gap

  \begin{enumerate}[leftmargin=1.5em,itemsep=0.5\baselineskip,topsep=0pt,parsep=0pt]
    \item
      \textbf{Deal} 5 cards each, clockwise from dealer's left, from a 52-card deck plus jokers.
      Turn one card face up to start the discard pile; the rest is placed face down as the pickup pile.
    \item
      \textbf{First} to play is the previous round's loser, or the youngest player.
      Any action of the first discard must be respected.
    \item
      \textbf{Play} a card matching the top discard's suit or rank.
      You may play several cards of the same rank at once (e.g. three 4s --- skip three turns).
      If you can't play, draw one card; you may play it right away.
    \item
      \textbf{Action cards} affect the next player:
      \settowidth{\actw}{\textbf{2 / 3:} draw 2 / 3}%
      \begin{itemize}[leftmargin=0pt,label={},itemsep=0pt,topsep=0.2\baselineskip,parsep=0pt]
        \item \actleft{\textbf{2 / 3:} draw 2 / 3}\textbf{4:} skip a turn
        \item \actleft{\textbf{A:} change the suit}\textbf{Joker:} is any card you name
        \item \textbf{K\Heart{} / K\Spade:} next / previous player draws 5
        \item \textbf{Q:} goes on any non-action card, and anything goes on it
        \item \textbf{J:} pass or demand a non-action rank, which every player must play
      \end{itemize}
    \item
      \textbf{Deflect} an action card with a matching one; penalties add up:
      \begin{itemize}[leftmargin=0pt,label={},itemsep=0pt,topsep=0.2\baselineskip,parsep=0pt]
        \item \textbf{2 / 3:} e.g.\ 2\Heart{} on 3\Heart: next player draws 5 (or deflects further)
        \item \textbf{K\Heart:} blocked by K\Diamondsuit{} / K\Club{}; deflected by 2\Heart, 3\Heart{} or K\Spade{}
        \item \textbf{J:} deflected only before a demanded card has been played
      \end{itemize}
    \item
      \textbf{Say ``Makao''} when you're down to one card, even partway through a multi-card play.
      If you forget, and someone notices when the next player plays, you draw 5.
    \item
      \textbf{Win} by emptying your hand first.
      Saying ``po makale'' (paw mah-KAH-leh \textipa{/pO ma"ka.lE/}) is a courtesy.
  \end{enumerate}

  \cardfooter{\footurlB}
}

\newcommand{\fourup}[1]{%
  \noindent\vbox to \paperheight{\offinterlineskip\cellrow{#1}\cellrow{#1}}}

% Dashed cut lines through the page centre; drawn on the front page only, so the
% back side prints clean. Must follow \fourup in the same paragraph.
\newcommand{\cutlines}{%
  \begin{tikzpicture}[remember picture,overlay]
    \draw[gray!60,dash pattern=on 3pt off 3pt,thin]
      (current page.north) -- (current page.south);
    \draw[gray!60,dash pattern=on 3pt off 3pt,thin]
      (current page.west) -- (current page.east);
  \end{tikzpicture}}

  \noindent\vbox to \paperheight{\offinterlineskip\scalebox{2}{\cell{#1}}\vss}}
